    <script>
        let dados = {
            inflaveis: JSON.parse(localStorage.getItem('inflaveis')) || [],
            mecanicos: JSON.parse(localStorage.getItem('mecanicos')) || [],
            jogos: JSON.parse(localStorage.getItem('jogos')) || [],
            veiculos: JSON.parse(localStorage.getItem('veiculos')) || []
        };

        // Auto-salvar a cada mudança
        function salvarDados() {
            Object.keys(dados).forEach(tipo => {
                localStorage.setItem(tipo, JSON.stringify(dados[tipo]));
            });
            mostrarMensagemSalvo();
        }

        function mostrarMensagemSalvo() {
            const status = document.getElementById('saveStatus');
            status.textContent = '✅ Dados salvos em ' + new Date().toLocaleTimeString();
            status.style.background = '#48bb78';
            setTimeout(() => {
                status.style.background = '#48bb78';
            }, 2000);
        }

        function showTab(tab) {
            document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
            document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
            event.target.classList.add('active');
            document.getElementById(tab).classList.add('active');
        }

        function adicionar(e, tipo) {
            e.preventDefault();
            const prefixo = tipo.substring(0, 3);
            const item = {
                id: Date.now(),
                nome: document.getElementById(prefixo + '_nome').value,
                data: document.getElementById(prefixo + '_data').value,
                obs: document.getElementById(prefixo + '_obs').value
            };
            if (tipo === 'inflaveis') item.tipo = document.getElementById('inf_tipo').value;
            
            dados[tipo].push(item);
            salvarDados();
            e.target.reset();
            atualizar();
            alert('✅ Cadastrado e salvo com sucesso!');
        }

        function excluir(tipo, id) {
            if (confirm('Excluir este item?')) {
                dados[tipo] = dados[tipo].filter(i => i.id !== id);
                salvarDados();
                atualizar();
            }
        }

        function calcularDias(data) {
            const diff = new Date() - new Date(data);
            return Math.floor(diff / (1000 * 60 * 60 * 24));
        }

        function atualizar() {
            document.getElementById('totalInflaveis').textContent = dados.inflaveis.length;
            document.getElementById('totalMecanicos').textContent = dados.mecanicos.length;
            document.getElementById('totalJogos').textContent = dados.jogos.length;
            document.getElementById('totalVeiculos').textContent = dados.veiculos.length;

            let alertas = 0;
            ['inflaveis', 'mecanicos', 'jogos', 'veiculos'].forEach(tipo => {
                const limite = tipo === 'veiculos' ? 40 : 30;
                const container = document.getElementById('lista_' + tipo);
                container.innerHTML = dados[tipo].map(item => {
                    const dias = calcularDias(item.data);
                    const badge = dias >= limite ? 'danger' : dias >= limite-5 ? 'warning' : 'success';
                    if (dias >= limite-5) alertas++;
                    return `
                        <div class="card">
                            <h3>${item.nome}</h3>
                            ${item.tipo ? '<p><strong>Tipo:</strong> ' + item.tipo + '</p>' : ''}
                            <p><strong>Última manutenção:</strong> ${new Date(item.data).toLocaleDateString()} (${dias} dias)</p>
                            ${item.obs ? '<p><strong>Obs:</strong> ' + item.obs + '</p>' : ''}
                            <span class="badge badge-${badge}">${dias >= limite ? '⚠️ Vencido' : dias >= limite-5 ? '⏰ Próximo' : '✅ OK'}</span>
                            <br><button class="btn btn-danger" onclick="excluir('${tipo}', ${item.id})">🗑️ Excluir</button>
                        </div>
                    `;
                }).join('');
            });
            document.getElementById('totalAlertas').textContent = alertas;
        }

        // SISTEMA DE BACKUP
        function fazerBackup() {
            const backup = {
                dados: dados,
                dataBackup: new Date().toISOString(),
                versao: '1.0'
            };
            const json = JSON.stringify(backup, null, 2);
            const blob = new Blob([json], {type: 'application/json'});
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'backup-manutencao-' + new Date().toISOString().split('T')[0] + '.json';
            a.click();
            
            localStorage.setItem('ultimoBackup', new Date().toISOString());
            atualizarInfoBackup();
            alert('✅ Backup baixado com sucesso!\n\nGuarde este arquivo em um local seguro (Google Drive, Email, etc.)');
        }

        function restaurarBackup() {
            const input = document.getElementById('fileInput');
            input.onchange = e => {
                const file = e.target.files[0];
                const reader = new FileReader();
                reader.onload = event => {
                    try {
                        const backup = JSON.parse(event.target.result);
                        if (confirm('⚠️ ATENÇÃO!\n\nRestaurar o backup vai SUBSTITUIR todos os dados atuais.\n\nTem certeza?')) {
                            dados = backup.dados;
                            salvarDados();
                            atualizar();
                            alert('✅ Backup restaurado com sucesso!');
                        }
                    } catch {
                        alert('❌ Erro ao restaurar backup!\n\nArquivo inválido.');
                    }
                };
                reader.readAsText(file);
            };
            input.click();
        }

        function copiarDados() {
            const textoCompleto = JSON.stringify(dados, null, 2);
            navigator.clipboard.writeText(textoCompleto).then(() => {
                alert('✅ Dados copiados!\n\nCole em um arquivo de texto e guarde como backup.');
            }).catch(() => {
                alert('📋 Copie manualmente:\n\n' + textoCompleto);
            });
        }

        function atualizarInfoBackup() {
            const ultimo = localStorage.getItem('ultimoBackup');
            if (ultimo) {
                const data = new Date(ultimo);
                document.getElementById('lastBackup').textContent = data.toLocaleString();
            }
        }

        // Inicializar
        document.addEventListener('DOMContentLoaded', () => {
            const hoje = new Date().toISOString().split('T')[0];
            document.querySelectorAll('input[type="date"]').forEach(i => i.value = hoje);
            atualizar();
            atualizarInfoBackup();
        });

        // Auto-backup a cada 24 horas
        setInterval(() => {
            const ultimo = localStorage.getItem('ultimoBackup');
            if (!ultimo || (Date.now() - new Date(ultimo).getTime()) > 86400000) {
                if (confirm('🔔 Lembrete de Backup!\n\nFaz mais de 24h desde o último backup.\n\nDeseja fazer backup agora?')) {
                    fazerBackup();
                }
            }
        }, 3600000); // Verifica a cada hora
    </script>
